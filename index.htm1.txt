<script>
    const products = [
        {id:1, name:"تی‌شرت مشکی Premium", price:320000, category:"مردانه", img:"https://picsum.photos/id/1015/600/400"},
        {id:2, name:"شلوار جین سفید", price:650000, category:"مردانه", img:"https://picsum.photos/id/1020/600/400"},
        {id:3, name:"هودی مشکی زمستانی", price:580000, category:"زمستانی", img:"https://picsum.photos/id/106/600/400"},
        {id:4, name:"پیراهن سفید مردانه", price:450000, category:"مردانه", img:"https://picsum.photos/id/201/600/400"},
        {id:5, name:"بلوز زنانه سفید", price:390000, category:"زنانه", img:"https://picsum.photos/id/64/600/400"},
        {id:6, name:"سویشرت مشکی", price:520000, category:"زمستانی", img:"https://picsum.photos/id/251/600/400"},
        {id:7, name:"تی‌شرت سفید اسپرت", price:280000, category:"زنانه", img:"https://picsum.photos/id/180/600/400"},
        {id:8, name:"جین بگ مردانه", price:720000, category:"مردانه", img:"https://picsum.photos/id/133/600/400"}
    ];

    let cart = JSON.parse(localStorage.getItem('perinCart')) || [];

    function renderProducts(list = products) {
        const grid = document.getElementById('productGrid');
        grid.innerHTML = '';
        if(list.length === 0) {
            grid.innerHTML = '<p style="text-align:center; grid-column:1/-1;">محصولی یافت نشد</p>';
            return;
        }
        list.forEach(p => {
            grid.innerHTML += `
                <div class="product-card">
                    <img src="${p.img}" class="product-img" alt="${p.name}">
                    <div class="product-info">
                        <h3>${p.name}</h3>
                        <div class="price">${p.price.toLocaleString('fa-IR')} تومان</div>
                        <button class="btn" style="width:100%;" onclick="addToCart(${p.id})">افزودن به سبد</button>
                    </div>
                </div>
            `;
        });
        // انیمیشن اسکرول
        setTimeout(() => {
            document.querySelectorAll('.product-card').forEach((card, i) => {
                setTimeout(() => card.classList.add('visible'), i * 100);
            });
        }, 100);
    }

    function filterProducts() {
        const search = document.getElementById('searchInput').value.toLowerCase();
        const category = document.getElementById('categoryFilter').value;
        const maxPrice = parseInt(document.getElementById('priceFilter').value);
        const filtered = products.filter(p => 
            p.name.toLowerCase().includes(search) &&
            (!category || p.category === category) &&
            p.price <= maxPrice
        );
        renderProducts(filtered);
    }

    function addToCart(id) {
        const product = products.find(p => p.id === id);
        cart.push(product);
        localStorage.setItem('perinCart', JSON.stringify(cart));
        updateCartCount();
        alert(product.name + ' به سبد اضافه شد ✅');
    }

    function updateCartCount() {
        document.getElementById('cartCount').textContent = cart.length;
    }

    function openCart() {
        const itemsDiv = document.getElementById('cartItems');
        let total = 0;
        let html = '';
        cart.forEach((item, i) => {
            total += item.price;
            html += `<div class="cart-item"><span>${item.name}</span><span>${item.price.toLocaleString('fa-IR')} ت <button class="remove-btn" onclick="removeFromCart(${i})">حذف</button></span></div>`;
        });
        itemsDiv.innerHTML = html || '<p>سبد خرید خالی است</p>';
        document.getElementById('cartTotal').textContent = 'مجموع: ' + total.toLocaleString('fa-IR') + ' تومان';
        openModal('cartModal');
    }

    function removeFromCart(index) {
        cart.splice(index, 1);
        localStorage.setItem('perinCart', JSON.stringify(cart));
        updateCartCount();
        openCart();
    }

    function checkout() {
        if(cart.length === 0) return alert('سبد خرید خالی است');
        alert('✅ در حال اتصال به درگاه پرداخت...\n(این بخش شبیه‌سازی است)');
        cart = [];
        localStorage.setItem('perinCart', JSON.stringify(cart));
        updateCartCount();
        closeModal('cartModal');
    }

    function doLogin() {
        const name = document.getElementById('userName').value;
        if(!name) return alert('نام کاربری را وارد کنید');
        alert('خوش آمدید ' + name + '!');
        closeModal('loginModal');
    }

    function doRegister() {
        const name = document.getElementById('userName').value;
        if(!name) return alert('نام کاربری را وارد کنید');
        alert('ثبت‌نام با موفقیت انجام شد!');
        closeModal('loginModal');
    }

    function openModal(id) { document.getElementById(id).style.display = 'flex'; }
    function closeModal(id) { document.getElementById(id).style.display = 'none'; }

    // انیمیشن اسکرول
    const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if(entry.isIntersecting) entry.target.classList.add('visible');
        });
    }, { threshold: 0.1 });

    renderProducts();
    updateCartCount();
</script>
